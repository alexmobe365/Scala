    ```python

        (1, 279.00, 0),
        (5, 18.50, 20),
        (2, 149.00, 5),
        (4, 89.90, 15),
    ]
    ```
    
    **Comprobación:** `calcular_importe(3, 59.90, 10)` debe devolver `161.73`.

# 1. Función para calcular el importe neto
def calcular_importe(unidades, precio_unitario, descuento_pct=0):
    """
    Calcula el importe neto de un pedido aplicando un descuento.
    
    Parámetros:
        unidades: número de unidades del pedido.
        precio_unitario: precio de cada unidad.
        descuento_pct: porcentaje de descuento. Por defecto es 0.
    
    Devuelve:
        El importe neto redondeado a dos decimales.
    """
    importe_bruto = unidades * precio_unitario
    descuento = importe_bruto * descuento_pct / 100
    importe_neto = importe_bruto - descuento
    
    return round(importe_neto, 2)


# 2. Función que devuelve el resumen del pedido
def resumen_pedido(unidades, precio_unitario, descuento_pct=0):
    """
    Devuelve un resumen de un pedido con importe bruto, ahorro e importe neto.
    
    Parámetros:
        unidades: número de unidades del pedido.
        precio_unitario: precio de cada unidad.
        descuento_pct: porcentaje de descuento. Por defecto es 0.
    
    Devuelve:
        Una tupla con (importe_bruto, ahorro, importe_neto).
    """
    importe_bruto = unidades * precio_unitario
    ahorro = importe_bruto * descuento_pct / 100
    importe_neto = importe_bruto - ahorro
    
    return (round(importe_bruto, 2), round(ahorro, 2), round(importe_neto, 2))


# 3. Llamada usando argumentos por palabra clave y desempaquetado
importe_bruto, ahorro, importe_neto = resumen_pedido(
    unidades=3,
    precio_unitario=59.90,
    descuento_pct=10
)

print("Importe bruto:", importe_bruto)
print("Ahorro:", ahorro)
print("Importe neto:", importe_neto)


# 4. Mostrar la ayuda de la función
help(calcular_importe)


# 5. Lista de pedidos
pedidos = [
    (3, 59.90, 10),
    (1, 279.00, 0),
    (5, 18.50, 20),
    (2, 149.00, 5),
    (4, 89.90, 15),
]

# Aplicar calcular_importe a cada pedido
importes = []

for pedido in pedidos:
    unidades, precio_unitario, descuento_pct = pedido
    importe = calcular_importe(unidades, precio_unitario, descuento_pct)
    importes.append(importe)

print("\nImportes:", importes)

# Usar max(), min(), sorted() y len()
print("Importe máximo:", max(importes))
print("Importe mínimo:", min(importes))
print("Importes ordenados:", sorted(importes))
print("Número de pedidos:", len(importes))

# 6. Crear resumen_por_region con un bucle

resumen_por_region = {}

for region in ventas["region"].unique():
    filtrado = ventas[ventas["region"] == region]

facturacion_groupby = (
